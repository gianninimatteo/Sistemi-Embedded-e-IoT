void setup() {
  Serial.begin(115200); //inizializza la comunicazione
  pinMode(LED_PIN, OUTPUT); //imposta il led come output
  digitalWrite(LED_PIN, LOW); //il led è spento all'inizio

  //connessione al wifi
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(1000);
    Serial.print(".");
  }
  Serial.println("\nWiFi connesso!");

  espClient.setCACert(root_ca);
  client.setServer(mqtt_server, mqtt_port);

  dht.begin();

  //connessione al broker mqtt
  Serial.println("Tentativo di connessione a MQTT...");
  if (client.connect("ESP32_Riscaldamento", mqtt_user, mqtt_password)) {
    Serial.println("Connesso al broker MQTT!");
  } else {
    Serial.print("Connessione fallita, stato: ");
    Serial.println(client.state());
  }
}

void loop() {
  if (!client.connected()) {
    Serial.println("MQTT disconnesso. Riconnessione...");
    if (client.connect("ESP32_Riscaldamento", mqtt_user, mqtt_password)) {
      Serial.println("Riconnesso a MQTT!");
    } else {
      Serial.println("Riconnessione fallita.");
      delay(1000);
      return;
    }
  }
  //lettura valori dal sensore
  float temperatura = dht.readTemperature();
  float umidita = dht.readHumidity();

  //controlla se la lettura dal sensore è valida
  if (isnan(temperatura) || isnan(umidita)) {
    Serial.println("Errore lettura DHT22!");
    return;
  }

  Serial.print("Temperatura: ");
  Serial.print(temperatura);
  Serial.print("°C, Umidità: ");
  Serial.print(umidita);
  Serial.println("%");

  //confronta il valore letto con la soglia 
  float soglia = 20.0;
  if (temperatura < soglia) {
    digitalWrite(LED_PIN, HIGH);
    client.publish(topic_riscaldamento, "ON");
  } else {
    digitalWrite(LED_PIN, LOW);
    client.publish(topic_riscaldamento, "OFF");
  }

  //converte temperatura e umidità in stringa per renderlo visibile su hivemq
  char tempString[8];
  dtostrf(temperatura, 1, 2, tempString);
  client.publish(topic_temperatura, tempString);

  char humString[8];
  dtostrf(umidita, 1, 2, humString);
  client.publish(topic_umidita, humString);

  client.loop();
  delay(2000);
}
